import React, { useState } from "react";
import { quranicVocab, otherVocab } from "./vocab";
import { PlayButton } from "./components/PlayButton";

function shuffle(array) {
  return [...array].sort(() => Math.random() - 0.5);
}

function VocabList({ vocab }) {
  return (
    <div>
      {vocab.map((w, i) => (
        <div key={i} style={{ marginBottom: "2.5rem", borderBottom: "1px #eee solid", paddingBottom: 16 }}>
          <h2 style={{ fontSize: "2em" }}>
            {w.arabic}
            <PlayButton text={w.arabic} />
          </h2>
          <p><b>Translation:</b> {w.translation}</p>
          <p><b>Example:</b> <em>{w.example}</em></p>
        </div>
      ))}
    </div>
  );
}

function Quiz({ vocab, onBack }) {
  const n = Math.min(vocab.length, 5);
  const quizWords = shuffle(vocab).slice(0, n);
  const [qIndex, setQIndex] = useState(0);
  const [score, setScore] = useState(0);
  const [selected, setSelected] = useState(null);
  const [showResult, setShowResult] = useState(false);

  const currentWord = quizWords[qIndex];
  const wrongAnswers = shuffle(vocab.filter(w => w.arabic !== currentWord.arabic)).slice(0, 3);
  const options = shuffle([currentWord, ...wrongAnswers]).map(w => w.translation);

  function handleOption(option) {
    setSelected(option);
    if (option === currentWord.translation) setScore(score + 1);
    setTimeout(() => {
      setSelected(null);
      if (qIndex + 1 < n) setQIndex(qIndex + 1);
      else setShowResult(true);
    }, 800);
  }

  if (showResult) {
    return (
      <div>
        <h2>Quiz Complete!</h2>
        <p style={{ fontSize: 24 }}>Score: {score} / {n}</p>
        <button onClick={onBack}>Back to Home</button>
        <button onClick={() => { setShowResult(false); setScore(0); setQIndex(0); }}>Retry Quiz</button>
      </div>
    );
  }
  return (
    <div>
      <button onClick={onBack}>Back</button>
      <h2>Question {qIndex + 1} / {n}</h2>
      <h3 style={{ fontSize: "2.5em" }}>
        {currentWord.arabic}
        <PlayButton text={currentWord.arabic} />
      </h3>
      <div style={{ margin: "1.1em 0" }}>
        {options.map(option => (
          <button
            key={option}
            style={{
              display: "block",
              margin: "0.5rem 0",
              padding: "0.7rem",
              background: selected
                ? option === currentWord.translation
                  ? "#b6f2a1"
                  : option === selected
                  ? "#ffbab7"
                  : "white"
                : "white",
              border: "1px solid #888",
              width: "100%",
              fontSize: "1.2em",
              cursor: selected ? "not-allowed" : "pointer"
            }}
            onClick={() => !selected && handleOption(option)}
            disabled={!!selected}
          >
            {option}
          </button>
        ))}
      </div>
      {selected && (
        <div style={{ marginTop: 12 }}>
          {selected === currentWord.translation ? (
            <span style={{ color: "green" }}>Correct!</span>
          ) : (
            <span style={{ color: "red" }}>
              Incorrect! Correct answer: {currentWord.translation}
            </span>
          )}
        </div>
      )}
    </div>
  );
}

export default function App() {
  const [page, setPage] = useState("home");
  const [quizType, setQuizType] = useState(null);

  return (
    <div style={{
      maxWidth: 600, margin: "auto", padding: 24,
      fontFamily: "system-ui, sans-serif"
    }}>
      <h1 style={{ textAlign: "center", color: "#257" }}>Learn Arabic Vocabulary</h1>
      {page === "home" && (
        <div style={{ display: "flex", flexDirection: "column", gap: 16, margin: "2em 0" }}>
          <button onClick={() => setPage("quranic")}>Quranic Vocabulary</button>
          <button onClick={() => setPage("other")}>Other Vocabulary</button>
          <button onClick={() => setPage("quizselector")}>Take a Quiz</button>
        </div>
      )}
      {page === "quranic" && (
        <div>
          <button onClick={() => setPage("home")}>Back</button>
          <VocabList vocab={quranicVocab} />
        </div>
      )}
      {page === "other" && (
        <div>
          <button onClick={() => setPage("home")}>Back</button>
          <VocabList vocab={otherVocab} />
        </div>
      )}
      {page === "quizselector" && (
        <div>
          <button onClick={() => setPage("home")}>Back</button>
          <h2 style={{ margin: "1em 0" }}>Select Quiz Category</h2>
          <button onClick={() => { setQuizType("quranic"); setPage("quiz"); }}>Quranic Vocabulary Quiz</button>
          <button onClick={() => { setQuizType("other"); setPage("quiz"); }}>Other Vocabulary Quiz</button>
        </div>
      )}
      {page === "quiz" && quizType === "quranic" && (
        <Quiz vocab={quranicVocab} onBack={() => setPage("home")} />
      )}
      {page === "quiz" && quizType === "other" && (
        <Quiz vocab={otherVocab} onBack={() => setPage("home")} />
      )}
    </div>
  );
}
